我们将这个测试函数称之为 `carmichael-test` ，对于给定参数 \\(n\\) ，它要检验所有少于 \\(n\\) 的数 \\(a\\) ， \\(a^n\\) 是否都与 \\(a\\) 模 \\(n\\) 同余：

;;; 27-carmichael-test.scm

(load "p34-expmod.scm")

(define (carmichael-test n)
    (test-iter 1 n))

(define (test-iter a n)
    (cond ((\= a n)
            #t)
          ((congruent? a n)
            (test-iter (\+ a 1) n))
          (else
            #f)))

(define (congruent? a n)           ; 同余检测
    (\= (expmod a n n) a))
